cmake_minimum_required(VERSION 3.16)
project(vla_control_validation)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_library(MOCK_COMPONENTS_LIBRARY mock_components REQUIRED PATHS /opt/ros/humble/lib)
add_library(vla_timing_mock SHARED timing_mock.cpp)
target_compile_features(vla_timing_mock PUBLIC cxx_std_17)
ament_target_dependencies(vla_timing_mock hardware_interface pluginlib rclcpp)
target_link_libraries(vla_timing_mock ${MOCK_COMPONENTS_LIBRARY})
pluginlib_export_plugin_description_file(hardware_interface plugins.xml)
install(TARGETS vla_timing_mock LIBRARY DESTINATION lib)
ament_package()
